How to fetch objects stored on the DB

Get all users

async function GetUsers() {
	try {
		const javascript_array = await fetch("/api/get_all?table=users", {
			method: "GET",
			// headers: {
			// 	"Authorization": `Bearer ${localStorage.getItem("token")}`
			// }
		});
		const users_json = await javascript_array.json();
		console.log(users_json);
	} catch (error) {
		console.error(error);
	}
}
